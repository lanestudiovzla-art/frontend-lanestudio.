let basePresupuestos = JSON.parse(localStorage.getItem('basePresupuestos')) || []; // 1. Guardar Presupuesto function guardarPresupuesto(cliente, items, totalUSD, tasaBCV) { const nuevoPresupuesto = { id: 'PRES-' + Math.floor(1000 + Math.random() * 9000), fecha: new Date().toLocaleString('es-VE', { hour12: true }), cliente: cliente, items: items, totalUSD: totalUSD, tasaBCV: tasaBCV, totalBS: totalUSD * tasaBCV, estado: 'En Espera' // 'En Espera' o 'Facturado' }; basePresupuestos.push(nuevoPresupuesto); localStorage.setItem('basePresupuestos', JSON.stringify(basePresupuestos)); alert(`Presupuesto ${nuevoPresupuesto.id} guardado con éxito "En Espera".`); } // 2. Cargar Presupuesto para revisión function cargarPresupuesto(id) { const pres = basePresupuestos.find(p => p.id === id); if (pres) { // Llenar formulario de la pantalla con los datos del presupuesto console.log("Cargando presupuesto:", pres); } } // 3. Convertir Presupuesto a Facturación function enviarAFacturacion(id) { const pres = basePresupuestos.find(p => p.id === id); if (pres) { pres.estado = 'Facturado'; localStorage.setItem('basePresupuestos', JSON.stringify(basePresupuestos)); // Redirigir/Cargar datos en el módulo de facturación alert(`Presupuesto ${id} enviado a Facturación con éxito.`); } }
